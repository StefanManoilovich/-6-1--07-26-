#include <iostream> // Подключаем библиотеку для ввода и вывода
using namespace std; // Разрешаем использовать cin и cout без std::

int main() { // Начало главной функции
    int n; // Создаём переменную для размера квадрата
    cin >> n; // Вводим размер квадрата

    for (int i = 0; i < n; i++) { // Перебираем строки
        for (int j = 0; j < n; j++) { // Перебираем столбцы
            if (i + j < n - 1) cout << 0; // Выводим 0 выше диагонали
            else if (i + j == n - 1) cout << 1; // Выводим 1 на диагонали
            else cout << 2; // Выводим 2 ниже диагонали

            if (j < n - 1) cout << " "; // Ставим пробел между числами
        } // Конец цикла по столбцам
        cout << endl; // Переходим на новую строку
    } // Конец цикла по строкам

    return 0; // Завершаем программу
} // Конец главной функции

Задание № 2 Способ № 1
#include <iostream> // Подключаем библиотеку ввода и вывода
using namespace std; // Разрешаем использовать cin и cout

int main() { // Начало программы
    int n, m, w, a[32][32] = {}; // Создаём поле и заполняем его нулями
    cin >> n >> m >> w; // Вводим число строк, столбцов и грибов

    for (int k = 0; k < w; k++) { // Перебираем все грибы
        int x, y; // Создаём переменные для координат гриба
        cin >> x >> y; // Считываем координаты
        a[x-1][y-1] = -1; // Отмечаем гриб в массиве
    } // Конец цикла ввода грибов

    for (int i = 0; i < n; i++) { // Перебираем строки поля
        for (int j = 0; j < m; j++) { // Перебираем столбцы поля
            if (a[i][j] == -1) continue; // Пропускаем клетки с грибами

            for (int x = i-1; x <= i+1; x++) { // Проверяем соседние строки
                for (int y = j-1; y <= j+1; y++) { // Проверяем соседние столбцы
                    if (x >= 0 && x < n && y >= 0 && y < m && a[x][y] == -1) // Проверяем границы и наличие гриба
                        a[i][j]++; // Увеличиваем количество соседних грибов
                } // Конец цикла по соседним столбцам
            } // Конец цикла по соседним строкам
        } // Конец цикла по столбцам
    } // Конец цикла по строкам

    for (int i = 0; i < n; i++) { // Перебираем строки для вывода
        for (int j = 0; j < m; j++) { // Перебираем столбцы для вывода
            if (a[i][j] == -1) cout << "*"; // Выводим звёздочку для гриба
            else cout << a[i][j]; // Выводим количество соседних грибов
            if (j < m-1) cout << " "; // Ставим пробел между клетками
        } // Конец цикла по столбцам
        cout << endl; // Переходим на следующую строку
    } // Конец цикла по строкам

    return 0; // Завершаем программу
} // Конец главной функции
Задание № 2 Способ № 2
#include <iostream> // Подключаем библиотеку ввода и вывода
using namespace std; // Подключаем пространство имён std

int main() { // Начало программы
    int n, m, w; // Объявляем размеры поля и число грибов
    char a[32][32]; // Создаём массив для карты
    int b[32][32] = {}; // Создаём массив для подсчёта грибов

    cin >> n >> m >> w; // Считываем размеры поля и число грибов

    for (int i = 0; i < n; i++) // Перебираем строки
        for (int j = 0; j < m; j++) // Перебираем столбцы
            a[i][j] = '.'; // Заполняем пустые клетки точками

    for (int k = 0; k < w; k++) { // Перебираем грибы
        int x, y; // Создаём переменные координат
        cin >> x >> y; // Считываем координаты гриба
        a[x-1][y-1] = '*'; // Отмечаем гриб звёздочкой
    } // Конец цикла ввода грибов

    for (int i = 0; i < n; i++) { // Перебираем строки
        for (int j = 0; j < m; j++) { // Перебираем столбцы
            if (a[i][j] == '*') continue; // Пропускаем клетки с грибами

            for (int x = i-1; x <= i+1; x++) { // Проверяем соседние строки
                for (int y = j-1; y <= j+1; y++) { // Проверяем соседние столбцы
                    if (x >= 0 && x < n && y >= 0 && y < m && a[x][y] == '*') // Проверяем границы и гриб
                        b[i][j]++; // Увеличиваем счётчик грибов
                } // Конец цикла по соседним столбцам
            } // Конец цикла по соседним строкам
        } // Конец цикла по столбцам
    } // Конец цикла по строкам

    for (int i = 0; i < n; i++) { // Перебираем строки для вывода
        for (int j = 0; j < m; j++) { // Перебираем столбцы для вывода
            if (a[i][j] == '*') cout << "*"; // Выводим гриб
            else cout << b[i][j]; // Выводим число соседних грибов
            if (j < m-1) cout << " "; // Ставим пробел между клетками
        } // Конец цикла по столбцам
        cout << endl; // Переходим на следующую строку
    } // Конец цикла по строкам

    return 0; // Завершаем программу
} // Конец главной функции
Задание № 3 Способ № 1
#include <iostream> // Подключаем библиотеку ввода и вывода
#include <string> // Подключаем библиотеку для работы со строками
using namespace std; // Подключаем пространство имён std

int main() { // Начало программы
    int x, n; // Объявляем начальное число и номер элемента
    cin >> x >> n; // Считываем числа x и n

    string s = to_string(x); // Преобразуем число в строку

    for (int k = 1; k < n; k++) { // Строим следующие элементы последовательности
        string r = ""; // Создаём пустую строку для результата
        int i = 0; // Начинаем с первого символа

        while (i < s.size()) { // Перебираем символы строки
            int j = i; // Запоминаем начало группы одинаковых цифр

            while (j < s.size() && s[j] == s[i]) // Ищем конец группы одинаковых цифр
                j++; // Переходим к следующему символу

            r += to_string(j-i) + s[i]; // Записываем количество цифр и саму цифру
            i = j; // Переходим к следующей группе
        } // Конец цикла по группам цифр

        s = r; // Заменяем строку новым элементом последовательности
    } // Конец цикла построения последовательности

    cout << s << endl; // Выводим нужный элемент
    return 0; // Завершаем программу
} // Конец главной функции
